SELECT City, Sales, Profit FROM "Super store"

SELECT City, Sales, Profit FROM "Super store" WHERE City = "Henderson"

-- Query 4: Sales 1000 se upar
SELECT City, Sales, Profit FROM "Super store" WHERE Sales > 1000

-- Query 5: City wise total sales
SELECT City, SUM(Sales) FROM "Super store" GROUP BY City ORDER BY SUM(Sales) DESC LIMIT 10
